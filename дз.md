using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;

public class QuestEntry
{
    public string QuestName { get; set; }
    public string Status { get; set; }  Active, Completed, Failed
    public string Description { get; set; }
}

public class AdventureJournal
{
    public string HeroName { get; set; }
    public ListQuestEntry Quests { get; set; } = new ListQuestEntry();
}

class Program
{
    static string path = Path.Combine(AppContext.BaseDirectory, journal.json);
    static AdventureJournal journal;

    static void Save()
    {
        var opts = new JsonSerializerOptions { WriteIndented = true };
        File.WriteAllText(path, JsonSerializer.Serialize(journal, opts));
    }

    static void Main()
    {
        if (File.Exists(path))
        {
            journal = JsonSerializer.DeserializeAdventureJournal(File.ReadAllText(path));
        }
        else
        {
            Console.Write(Введите имя героя );
            journal = new AdventureJournal
            {
                HeroName = Console.ReadLine(),
                Quests = new ListQuestEntry()
            };
            Save();
        }

        while (true)
        {
            Console.WriteLine($n=== Журнал приключений ({journal.HeroName}) ===);
            Console.WriteLine(1. Добавить квест);
            Console.WriteLine(2. Завершить квест);
            Console.WriteLine(3. Провалить квест);
            Console.WriteLine(4. Показать все квесты);
            Console.WriteLine(5. Показать только активные квесты);
            Console.WriteLine(6. Выход);
            Console.Write(Выбор );
            string choice = Console.ReadLine();

            switch (choice)
            {
                case 1 AddQuest(); break;
                case 2 ChangeStatus(Completed); break;
                case 3 ChangeStatus(Failed); break;
                case 4 ShowAll(); break;
                case 5 ShowActive(); break;
                case 6 return;
            }
        }
    }

    static void AddQuest()
    {
        Console.Write(Название квеста );
        string name = Console.ReadLine();
        Console.Write(Описание );
        string desc = Console.ReadLine();
        journal.Quests.Add(new QuestEntry { QuestName = name, Description = desc, Status = Active });
        Save();
        Console.WriteLine(Квест добавлен.);
    }

    static void ChangeStatus(string newStatus)
    {
        var active = journal.Quests.FindAll(q = q.Status == Active);
        if (active.Count == 0) { Console.WriteLine(Нет активных квестов.); return; }

        for (int i = 0; i  active.Count; i++)
            Console.WriteLine(${i + 1}. {active[i].QuestName});

        Console.Write(Выберите номер );
        if (!int.TryParse(Console.ReadLine(), out int n)  n  1  n  active.Count)
        { Console.WriteLine(Неверный номер.); return; }

        active[n - 1].Status = newStatus;
        Save();
        Console.WriteLine($Статус изменён на {newStatus}.);
    }

    static void ShowAll()
    {
        if (journal.Quests.Count == 0) { Console.WriteLine(Квестов нет.); return; }
        foreach (var q in journal.Quests)
            Console.WriteLine($[{q.Status}] {q.QuestName} — {q.Description});
    }

    static void ShowActive()
    {
        var active = journal.Quests.FindAll(q = q.Status == Active);
        if (active.Count == 0) { Console.WriteLine(Активных квестов нет.); return; }
        foreach (var q in active)
            Console.WriteLine(${q.QuestName} — {q.Description});
    }
}

using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;

public class Monster
{
    public string Name { get; set; }
    public int Health { get; set; }
    public int Damage { get; set; }
    public string Type { get; set; }
}

class Program
{
    static void Main()
    {
        string csvPath = Path.Combine(AppContext.BaseDirectory, "monsters.csv");
        string jsonPath = Path.Combine(AppContext.BaseDirectory, "monsters.json");

        string[] lines = File.ReadAllLines(csvPath);
        var monsters = new List<Monster>();

        for (int i = 1; i < lines.Length; i++) // пропускаем заголовок
        {
            if (string.IsNullOrWhiteSpace(lines[i])) continue;
            string[] p = lines[i].Split(';');
            monsters.Add(new Monster
            {
                Name = p[0],
                Health = int.Parse(p[1]),
                Damage = int.Parse(p[2]),
                Type = p[3]
            });
        }

        var opts = new JsonSerializerOptions { WriteIndented = true };
        File.WriteAllText(jsonPath, JsonSerializer.Serialize(monsters, opts));

        var loaded = JsonSerializer.Deserialize<List<Monster>>(File.ReadAllText(jsonPath));

        Console.WriteLine("=== Бестиарий ===");
        foreach (var m in loaded)
            Console.WriteLine($"[{m.Type}] {m.Name} — HP: {m.Health}, DMG: {m.Damage}");

        Monster maxHp = loaded[0], maxDmg = loaded[0];
        foreach (var m in loaded)
        {
            if (m.Health > maxHp.Health) maxHp = m;
            if (m.Damage > maxDmg.Damage) maxDmg = m;
        }

        Console.WriteLine($"\nВсего монстров: {loaded.Count}");
        Console.WriteLine($"Макс. HP: {maxHp.Name} ({maxHp.Health})");
        Console.WriteLine($"Макс. DMG: {maxDmg.Name} ({maxDmg.Damage})");
    }
}

using System;
using System.IO;
using System.Text.Json;

public class PlayerProfile
{
    public string Nickname { get; set; }
    public int GamesPlayed { get; set; }
    public int Wins { get; set; }
    public double WinRate { get; set; }
}

class Program
{
    static string logPath = Path.Combine(AppContext.BaseDirectory, "actions.log");

    static string ProfilePath(string nick) =>
        Path.Combine(AppContext.BaseDirectory, $"profile_{nick}.json");

    static void Log(string text)
    {
        string stamp = DateTime.Now.ToString("dd.MM.yyyy HH:mm");
        File.AppendAllText(logPath, $"[{stamp}] {text}\n");
    }

    static void Main()
    {
        var opts = new JsonSerializerOptions { WriteIndented = true };

        while (true)
        {
            Console.WriteLine("\n=== Управление профилями ===");
            Console.WriteLine("1. Создать профиль");
            Console.WriteLine("2. Записать результат игры");
            Console.WriteLine("3. Показать профиль");
            Console.WriteLine("4. Показать лог действий");
            Console.WriteLine("5. Выход");
            Console.Write("Выбор: ");
            string choice = Console.ReadLine();

            switch (choice)
            {
                case "1":
                {
                    Console.Write("Никнейм: ");
                    string nick = Console.ReadLine();
                    string p = ProfilePath(nick);
                    if (File.Exists(p)) { Console.WriteLine("Профиль уже существует!"); break; }

                    var prof = new PlayerProfile { Nickname = nick, GamesPlayed = 0, Wins = 0, WinRate = 0 };
                    File.WriteAllText(p, JsonSerializer.Serialize(prof, opts));
                    Log($"Создан профиль: {nick}");
                    Console.WriteLine("Профиль создан.");
                    break;
                }
                case "2":
                {
                    Console.Write("Никнейм: ");
                    string nick = Console.ReadLine();
                    string p = ProfilePath(nick);
                    if (!File.Exists(p)) { Console.WriteLine("Профиль не найден"); break; }

                    Console.Write("Результат (win/loss): ");
                    string res = Console.ReadLine();

                    var prof = JsonSerializer.Deserialize<PlayerProfile>(File.ReadAllText(p));
                    prof.GamesPlayed++;
                    if (res == "win") prof.Wins++;
                    prof.WinRate = prof.Wins * 100.0 / prof.GamesPlayed;
                    File.WriteAllText(p, JsonSerializer.Serialize(prof, opts));

                    Log($"{nick}: {res} (Всего игр: {prof.GamesPlayed}, Побед: {prof.Wins}, WinRate: {prof.WinRate:F1}%)");
                    Console.WriteLine("Результат записан.");
                    break;
                }
                case "3":
                {
                    Console.Write("Никнейм: ");
                    string nick = Console.ReadLine();
                    string p = ProfilePath(nick);
                    if (!File.Exists(p)) { Console.WriteLine("Профиль не найден"); break; }

                    var prof = JsonSerializer.Deserialize<PlayerProfile>(File.ReadAllText(p));
                    Console.WriteLine($"Никнейм: {prof.Nickname}");
                    Console.WriteLine($"Игр: {prof.GamesPlayed}");
                    Console.WriteLine($"Побед: {prof.Wins}");
                    Console.WriteLine($"WinRate: {prof.WinRate:F1}%");
                    break;
                }
                case "4":
                {
                    if (File.Exists(logPath)) Console.WriteLine(File.ReadAllText(logPath));
                    else Console.WriteLine("Лог пуст.");
                    break;
                }
                case "5": return;
            }
        }
    }
}
