```csharp
using System;
using System.Collections.Generic;
using System.Linq;

namespace StatusEffects
{
    public interface IStatusEffect
    {
        string Name { get; }
        int Duration { get; }
        bool CanStack { get; }
        string StackKey { get; } // Для проверки стакания (два эффекта одинаковые, если совпадают ключи)

        void Apply(Character character);
        void Tick(); 
    }

    public class Character
    {
        public string Name { get; }
        public int HP { get; private set; }
        public int MaxHP { get; }
        public int Armor { get; }
        public bool IsFrozen { get; set; } // Флаг для заморозки
        public List<IStatusEffect> ActiveEffects { get; } = new List<IStatusEffect>();

        public Character(string name, int maxHp, int armor)
        {
            Name = name;
            MaxHP = maxHp;
            HP = maxHp;
            Armor = armor;
        }

        public void TakeDamage(int amount, bool ignoreArmor = false)
        {
            int actualDamage = ignoreArmor ? amount : Math.Max(1, amount - Armor);
            HP = Math.Max(0, HP - actualDamage);
            Console.WriteLine($"  {Name} получает {actualDamage} урона (HP: {HP}/{MaxHP})");
        }

        public void Heal(int amount)
        {
            HP = Math.Min(MaxHP, HP + amount);
            Console.WriteLine($"  {Name} лечится на {amount} HP (HP: {HP}/{MaxHP})");
        }

        public void ApplyEffect(IStatusEffect effect)
        {
            var existing = ActiveEffects.FirstOrDefault(e => e.StackKey == effect.StackKey);

            if (existing != null)
            {
                if (existing.CanStack)
                {
                   
                    for (int i = 0; i < effect.Duration; i++) existing.Tick();
                    Console.WriteLine($"  Эффект {effect.Name} продлен. Новая длительность: {existing.Duration}");
                }
                else
                {
                    Console.WriteLine($"  Эффект {effect.Name} уже висит и не стакается. Длительность обновлена.");
                   
                    ActiveEffects.Remove(existing);
                    ActiveEffects.Add(effect);
                }
            }
            else
            {
                ActiveEffects.Add(effect);
                Console.WriteLine($"  Наложен эффект: {effect.Name} ({effect.Duration} ходов)");
            }
        }

        public void ProcessTurn()
        {
            Console.WriteLine($"\n--- Ход персонажа {Name} ---");
            IsFrozen = false; // Сбрасываем флаг заморозки в начале хода

            foreach (var effect in ActiveEffects.ToList())
            {
                effect.Apply(this);
            }

            foreach (var effect in ActiveEffects.ToList())
            {
                effect.Tick();
                if (effect.Duration <= 0)
                {
                    Console.WriteLine($"  Эффект {effect.Name} закончился.");
                    ActiveEffects.Remove(effect);
                }
            }
        }

        public void ShowStatus()
        {
            Console.WriteLine($"Статус {Name}: HP {HP}/{MaxHP} | Броня {Armor}");
            if (IsFrozen) Console.WriteLine("  [ЗАМОРОЖЕН] Не может действовать!");
            
            if (ActiveEffects.Count > 0)
            {
                Console.WriteLine("  Активные эффекты: " + string.Join(", ", ActiveEffects.Select(e => $"{e.Name}({e.Duration})")));
            }
        }
    }

    public class PoisonEffect : IStatusEffect
    {
        public string Name => "Отравление";
        public int Duration { get; private set; } = 3;
        public bool CanStack => true;
        public string StackKey => "Poison";
        private readonly int _damagePerTurn;

        public PoisonEffect(int damagePerTurn = 5) { _damagePerTurn = damagePerTurn; }

        public void Apply(Character character)
        {
            Console.WriteLine($"  {Name} наносит урон, игнорируя броню.");
            character.TakeDamage(_damagePerTurn, ignoreArmor: true);
        }
        public void Tick() => Duration--;
    }

    public class BurnEffect : IStatusEffect
    {
        public string Name => "Горение";
        public int Duration { get; private set; } = 2;
        public bool CanStack => false;
        public string StackKey => "Burn";
        private readonly int _damagePerTurn;

        public BurnEffect(int damagePerTurn = 10) { _damagePerTurn = damagePerTurn; }

        public void Apply(Character character)
        {
            Console.WriteLine($"  {Name} наносит урон (броня снижает).");
            character.TakeDamage(_damagePerTurn);
        }
        public void Tick() => Duration--;
    }

    public class RegenerationEffect : IStatusEffect
    {
        public string Name => "Регенерация";
        public int Duration { get; private set; } = 4;
        public bool CanStack => false;
        public string StackKey => "Regen";
        private readonly int _healAmount;

        public RegenerationEffect(int healAmount = 8) { _healAmount = healAmount; }

        public void Apply(Character character)
        {
            Console.WriteLine($"  {Name} лечит.");
            character.Heal(_healAmount);
        }
        public void Tick() => Duration--;
    }

    public class FreezeEffect : IStatusEffect
    {
        public string Name => "Заморозка";
        public int Duration { get; private set; } = 1;
        public bool CanStack => false;
        public string StackKey => "Freeze";

        public void Apply(Character character)
        {
            Console.WriteLine($"  {Name} замораживает персонажа!");
            character.IsFrozen = true;
        }
        public void Tick() => Duration--;
    }

    class Program
    {
        static void Main()
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;
            
            var hero = new Character("Рыцарь", 100, 5);
            hero.ShowStatus();

            Console.WriteLine("\nНакладываем эффекты:");
            hero.ApplyEffect(new PoisonEffect(6));
            hero.ApplyEffect(new BurnEffect(10));
            hero.ApplyEffect(new RegenerationEffect(8));
            
            hero.ApplyEffect(new PoisonEffect(6)); 
            hero.ApplyEffect(new BurnEffect(10));  
            hero.ApplyEffect(new FreezeEffect());

            for (int turn = 1; turn <= 4; turn++)
            {
                Console.WriteLine($"\n== ХОД {turn} ==");
                hero.ProcessTurn();
                hero.ShowStatus();
            }
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace EnvironmentInteraction
{
    public interface IEntityVisitor
    {
        string Name { get; }
        int HP { get; set; }
        
        void VisitChest(Chest chest);
        void VisitTrap(Trap trap);
        void VisitCampfire(Campfire campfire);
    }

    public interface IEnvironmentObject
    {
        string Name { get; }
        // Двойная диспетчеризация: объект просит сущность обработать себя
        void Accept(IEntityVisitor visitor);
    }

    public interface IEntity : IEntityVisitor
    {
        void InteractWith(IEnvironmentObject obj);
    }

  
    
    public class Chest : IEnvironmentObject
    {
        public string Name => "Сундук";
        public bool IsOpen { get; private set; }
        public int Gold { get; private set; } = 100;

        public void Accept(IEntityVisitor visitor)
        {
            visitor.VisitChest(this);
        }

        public void Open()
        {
            if (!IsOpen)
            {
                IsOpen = true;
                Console.WriteLine($"  >> {Name} открыт! Внутри {Gold} золота.");
            }
        }

        public void Break()
        {
            Console.WriteLine($"  >> {Name} разбит! Золото рассыпалось.");
            IsOpen = false;
            Gold = 0;
        }

        public void TakeGold()
        {
            if (IsOpen && Gold > 0)
            {
                Console.WriteLine($"  >> Забрано {Gold} золота.");
                Gold = 0;
            }
        }
    }

    public class Trap : IEnvironmentObject
    {
        public string Name => "Ловушка";
        public bool IsDisarmed { get; private set; }
        public int Damage { get; } = 20;

        public void Accept(IEntityVisitor visitor)
        {
            visitor.VisitTrap(this);
        }

        public void Disarm()
        {
            if (!IsDisarmed)
            {
                IsDisarmed = true;
                Console.WriteLine($"  >> {Name} обезврежена.");
            }
        }

        public void Trigger(IEntityVisitor target)
        {
            if (!IsDisarmed)
            {
                Console.WriteLine($"  >> {Name} срабатывает! {target.Name} получает {Damage} урона.");
                target.HP -= Damage;
                IsDisarmed = true;
            }
        }
    }

    public class Campfire : IEnvironmentObject
    {
        public string Name => "Костёр";
        public bool IsLit { get; private set; } = true;

        public void Accept(IEntityVisitor visitor)
        {
            visitor.VisitCampfire(this);
        }

        public void Ignite()
        {
            if (!IsLit)
            {
                IsLit = true;
                Console.WriteLine($"  >> {Name} разожжён!");
            }
            else Console.WriteLine($"  >> {Name} уже горит.");
        }

        public void Extinguish()
        {
            if (IsLit)
            {
                IsLit = false;
                Console.WriteLine($"  >> {Name} потушен.");
            }
        }
    }

 

    public abstract class BaseEntity : IEntity
    {
        public string Name { get; }
        public int HP { get; set; }

        protected BaseEntity(string name, int hp)
        {
            Name = name;
            HP = hp;
        }

        public void InteractWith(IEnvironmentObject obj)
        {
            Console.WriteLine($"{Name} взаимодействует с {obj.Name}:");
            obj.Accept(this); // Запускаем двойную диспетчеризацию
        }

        public abstract void VisitChest(Chest chest);
        public abstract void VisitTrap(Trap trap);
        public abstract void VisitCampfire(Campfire campfire);
    }

    public class Player : BaseEntity
    {
        public Player() : base("Игрок", 100) { }

        public override void VisitChest(Chest chest)
        {
            chest.Open();
            chest.TakeGold();
        }

        public override void VisitTrap(Trap trap)
        {
            trap.Disarm();
        }

        public override void VisitCampfire(Campfire campfire)
        {
            campfire.Ignite();
        }
    }

    public class Enemy : BaseEntity
    {
        public Enemy() : base("Враг", 100) { }

        public override void VisitChest(Chest chest)
        {
            Console.WriteLine($"  >> {Name} бьёт сундук.");
            chest.Break();
        }

        public override void VisitTrap(Trap trap)
        {
            trap.Trigger(this);
        }

        public override void VisitCampfire(Campfire campfire)
        {
            campfire.Extinguish();
        }
    }

    public class Ally : BaseEntity
    {
        public Ally() : base("Союзник", 80) { }

        public override void VisitChest(Chest chest)
        {
            Console.WriteLine($"  >> {Name} игнорирует сундук.");
        }

        public override void VisitTrap(Trap trap)
        {
            Console.WriteLine($"  >> {Name} обходит ловушку.");
        }

        public override void VisitCampfire(Campfire campfire)
        {
            Console.WriteLine($"  >> {Name} греется у костра.");
        }
    }

    class Program
    {
        static void Main()
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;

            var entities = new List<IEntity>
            {
                new Player(),
                new Enemy(),
                new Ally()
            };

            var objects = new List<IEnvironmentObject>
            {
                new Chest(),
                new Trap(),
                new Campfire()
            };

            // Проводим все возможные комбинации взаимодействий
            foreach (var entity in entities)
            {
                Console.WriteLine($"\n=================== {entity.Name.ToUpper()} ===================");
                foreach (var obj in objects)
                {
                
                    IEnvironmentObject freshObj = obj switch
                    {
                        Chest => new Chest(),
                        Trap => new Trap(),
                        Campfire => new Campfire(),
                        _ => obj
                    };
                    
                    entity.InteractWith(freshObj);
                    Console.WriteLine($"  [HP сущности: {entity.HP}]");
                }
            }
        }
    }
}
```
